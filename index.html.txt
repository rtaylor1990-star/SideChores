    function playAudioTone(type) {
      try {
        const ctx = new (window.AudioContext || window.webkitAudioContext)();
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.connect(gain);
        gain.connect(ctx.destination);

        if (type === 'complete') {
          osc.type = 'sine';
          osc.frequency.setValueAtTime(523.25, ctx.currentTime);
          osc.frequency.exponentialRampToValueAtTime(659.25, ctx.currentTime + 0.1);
          osc.frequency.exponentialRampToValueAtTime(783.99, ctx.currentTime + 0.2);
          gain.gain.setValueAtTime(0.3, ctx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.4);
          osc.start(ctx.currentTime);
          osc.stop(ctx.currentTime + 0.4);
        } else if (type === 'uncomplete') {
          osc.type = 'triangle';
          osc.frequency.setValueAtTime(300, ctx.currentTime);
          osc.frequency.exponentialRampToValueAtTime(150, ctx.currentTime + 0.15);
          gain.gain.setValueAtTime(0.3, ctx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.2);
          osc.start(ctx.currentTime);
          osc.stop(ctx.currentTime + 0.2);
        } else if (type === 'redeem') {
          osc.type = 'square';
          osc.frequency.setValueAtTime(987.77, ctx.currentTime);
          osc.frequency.setValueAtTime(1318.51, ctx.currentTime + 0.08);
          gain.gain.setValueAtTime(0.2, ctx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.3);
          osc.start(ctx.currentTime);
          osc.stop(ctx.currentTime + 0.3);
        } else if (type === 'error') {
          osc.type = 'sawtooth';
          osc.frequency.setValueAtTime(150, ctx.currentTime);
          gain.gain.setValueAtTime(0.3, ctx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.25);
          osc.start(ctx.currentTime);
          osc.stop(ctx.currentTime + 0.25);
        }
      } catch (e) {}
    }

    // --- TASK INTERACTIONS (KID VIEW) ---
    function toggleTaskCompletion(taskId) {
      const currentChild = getActiveChild();
      const task = state.tasks.find(t => t.id === taskId);
      if (!task) return;

      if (!task.completedBy) task.completedBy = {};

      const isCompleted = !!task.completedBy[currentChild.id];

      if (!isCompleted) {
        task.completedBy[currentChild.id] = true;
        currentChild.balance += task.mins;
        playAudioTone('complete');
        triggerConfetti();
        showToast(`Awesome ${currentChild.name}! 🎉`, `Completed "${task.title}" and earned +${task.mins} minutes!`, "fa-star");
      } else {
        delete task.completedBy[currentChild.id];
        currentChild.balance = Math.max(0, currentChild.balance - task.mins);
        playAudioTone('uncomplete');
        showToast("Task Unchecked", `Subtracted ${task.mins} minutes from ${currentChild.name}.`, "fa-rotate-left");
      }

      saveState();
    }

    function requestRedeem(mins) {
      const currentChild = getActiveChild();

      if (currentChild.balance < mins) {
        playAudioTone('error');
        showToast("Not Enough Minutes!", `${currentChild.name} needs ${mins} mins, but only has ${currentChild.balance} mins available.`, "fa-circle-exclamation");
        return;

      const newRequest = {
        id: Date.now().toString(),
        childId: currentChild.id,
        mins: mins,
        timestamp: Date.now()
      };

      state.pendingRequests.push(newRequest);
      playAudioTone('redeem');
      showToast("Request Sent! ⏳", `Asked parent to spend ${mins} minutes for ${currentChild.name}.`, "fa-hourglass-half");
      saveState();
    }

    function deleteChildProfile(childId) {
      if (state.children.length <= 1) {
        showToast("Cannot Delete", "You must keep at least one child profile.", "fa-triangle-exclamation");
        return;
      }

      const child = state.children.find(c => c.id === childId);
      if (!child) return;

      if (confirm(`Are you sure you want to delete profile for ${child.name}?`)) {
        // Remove child profile
        state.children = state.children.filter(c => c.id !== childId);
        
        // Clean up pending requests associated with this child
        state.pendingRequests = state.pendingRequests.filter(r => r.childId !== childId);
        
        // Clean up task completions for this child across all tasks
        state.tasks.forEach(task => {
          if (task.completedBy && task.completedBy[childId]) {
            delete task.completedBy[childId];
          }
        });

        // Clean up specific tasks assigned only to this child
        state.tasks = state.tasks.filter(t => t.childId !== childId);

        // If active child was deleted, fallback to the first available child
        if (state.activeChildId === childId) {
          state.activeChildId = state.children.length > 0 ? state.children[0].id : null;
        }

        playAudioTone('uncomplete');
        showToast("Profile Removed", `Deleted profile for ${child.name}.`, "fa-trash");
        saveState();
      }
    }

    function approveRedeem(requestId) {
      const index = state.pendingRequests.findIndex(r => r.id === requestId);
      if (index === -1) return;

      const req = state.pendingRequests[index];
      const reqChild = state.children.find(c => c.id === req.childId);

      if (reqChild && reqChild.balance >= req.mins) {
        reqChild.balance -= req.mins;
        state.pendingRequests.splice(index, 1);
        playAudioTone('complete');
        showToast("Request Approved! 👍", `Deducted ${req.mins} mins from ${reqChild.name}'s balance.`, "fa-check");
      } else {
        state.pendingRequests.splice(index, 1);
        playAudioTone('error');
        showToast("Insufficient Balance", "Child balance was lower than requested amount.", "fa-xmark");
      }
      saveState();
    }

    function denyRedeem(requestId) {
      state.pendingRequests = state.pendingRequests.filter(r => r.id !== requestId);
      playAudioTone('uncomplete');
      showToast("Request Denied", "Tech time request was rejected.", "fa-xmark");
      saveState();
    }

    function adjustBalance(amount) {
      const currentChild = getActiveChild();
      currentChild.balance = Math.max(0, currentChild.balance + amount);
      playAudioTone('complete');
      saveState();
    }

    function resetBalanceToZero() {
      const currentChild = getActiveChild();
      if (confirm(`Reset tech time balance for ${currentChild.name} to 0 minutes?`)) {
        currentChild.balance = 0;
        saveState();
      }
    }

    function resetDailyChores(allKids = false) {
      if (allKids) {
        state.tasks.forEach(t => t.completedBy = {});
        showToast("All Tasks Reset!", "Unchecked daily chores for all children.", "fa-sun");
      } else {
        const currentChild = getActiveChild();
        state.tasks.forEach(t => {
          if (t.completedBy) delete t.completedBy[currentChild.id];
        });
        showToast("Reset Complete!", `Unchecked daily chores for ${currentChild.name}.`, "fa-sun");
      }
      saveState();
    }

    function handleAddTask(event) {
      event.preventDefault();
      const titleInput = document.getElementById('newTaskTitle');
      const minsInput = document.getElementById('newTaskMins');
      const emojiInput = document.getElementById('newTaskEmoji');
      const assignInput = document.getElementById('newTaskAssignee');

      const title = titleInput.value.trim();
      const mins = parseInt(minsInput.value, 10);
      const emoji = emojiInput.value;
      const childId = assignInput.value;

      if (!title || isNaN(mins)) return;

      const newTask = {
        id: Date.now().toString(),
        childId: childId,
        title: title,
        mins: mins,
        emoji: emoji,
        completedBy: {}
      };

      state.tasks.push(newTask);
      titleInput.value = '';
      minsInput.value = '10';

      playAudioTone('complete');
      showToast("Chore Added", `Added "${title}" worth ${mins} mins.`, "fa-plus");
      saveState();
    }

    function updateTaskTitle(taskId, newTitle) {
      const task = state.tasks.find(t => t.id === taskId);
      if (task && newTitle.trim()) {
        task.title = newTitle.trim();
        saveState();
      }
    }

    function updateTaskMins(taskId, newMins) {
      const task = state.tasks.find(t => t.id === taskId);
      const val = parseInt(newMins, 10);
      if (task && !isNaN(val) && val > 0) {
        task.mins = val;
        saveState();
      }
    }

    function updateTaskAssignee(taskId, childId) {
      const task = state.tasks.find(t => t.id === taskId);
      if (task) {
        task.childId = childId;
        saveState();
      }
    }

    function deleteTask(taskId) {
      if (confirm("Are you sure you want to delete this chore definition?")) {
        state.tasks = state.tasks.filter(t => t.id !== taskId);
        saveState();
      }
    }

    // --- PIN MODAL & SECURITY ---
    function openPinModal(changingPin = false) {
      isChangingPinMode = changingPin;
      enteredPin = "";
      updatePinDots();
      document.getElementById('pinErrorMsg').classList.add('hidden');

      const titleEl = document.getElementById('pinModalTitle');
      const descEl = document.getElementById('pinModalDesc');

      if (changingPin) {
        titleEl.innerText = "Set New PIN";
        descEl.innerText = "Enter a new 4-digit PIN for parent gate.";
      } else {
        titleEl.innerText = "Parent Verification";
        descEl.innerText = "Enter your 4-digit PIN to enter Admin mode.";
      }

      document.getElementById('pinModal').classList.remove('hidden');
    }

    function closePinModal() {
      document.getElementById('pinModal').classList.add('hidden');
      enteredPin = "";
    }

    function inputPinNum(num) {
      if (enteredPin.length < 4) {
        enteredPin += num.toString();
        updatePinDots();
      }

      if (enteredPin.length === 4) {
        setTimeout(verifyPin, 100);
      }
    }

    function deleteLastPinDigit() {
      if (enteredPin.length > 0) {
        enteredPin = enteredPin.slice(0, -1);
        updatePinDots();
      }
    }

    function clearPin() {
      enteredPin = "";
      updatePinDots();
    }

    function updatePinDots() {
      const dots = document.querySelectorAll('.pin-dot');
      dots.forEach((dot, index) => {
        if (index < enteredPin.length) {
          dot.className = "pin-dot w-4 h-4 rounded-full bg-purple-600 border-2 border-purple-600 scale-110 transition-all";
        } else {
          dot.className = "pin-dot w-4 h-4 rounded-full border-2 border-slate-300 transition-all";
        }
      });
    }

    function verifyPin() {
      if (isChangingPinMode) {
        state.pin = enteredPin;
        saveState();
        closePinModal();
        playAudioTone('complete');
        showToast("PIN Updated!", "Your new 4-digit parent PIN has been saved.", "fa-key");
      } else {
        if (enteredPin === state.pin) {
          state.isParentUnlocked = true;
          closePinModal();
          switchView('parent');
          playAudioTone('complete');
        } else {
          playAudioTone('error');
          document.getElementById('pinErrorMsg').classList.remove('hidden');
          enteredPin = "";
          updatePinDots();
        }
      }
    }

        confetti({
          particleCount: 80,
          spread: 70,
          origin: { y: 0.6 }
        });

    }

    // --- TOAST NOTIFICATION UTILITY ---
    let toastTimeout;
    function showToast(title, message, iconClass = 'fa-bell') {
      const toast = document.getElementById('toastNotification');
      document.getElementById('toastTitle').innerText = title;
      document.getElementById('toastMessage').innerText = message;
      document.getElementById('toastIcon').innerHTML = `<i class="fa-solid ${iconClass}"></i>`;

      toast.classList.remove('hidden', 'opacity-0', 'translate-y-4');
      toast.classList.add('opacity-100', 'translate-y-0');

      clearTimeout(toastTimeout);
      toastTimeout = setTimeout(() => {
        toast.classList.remove('opacity-100', 'translate-y-0');
        toast.classList.add('opacity-0', 'translate-y-4');
        setTimeout(() => toast.classList.add('hidden'), 300);
      }, 3500);
    }

    function escapeHtml(str) {
      return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");
    }
